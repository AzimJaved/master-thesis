\section{Introduction}

\begin{frame}{In-Memory Computing \cite{rossenbach2025}}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{3pt}
        \item Various technologies exist, with varying tradeoffs:
              \begin{itemize}
                \item Chip complexity and density
                \item Energy consumption
                \item Speed and reliability
              \end{itemize}
        \item This talk focuses on \emph{memristive} hardware, in particular \emph{resistive RAM (ReRAM)}:
              \begin{itemize}
                \item Key Benefits: low energy consumption, non volatility
              \end{itemize}
      \end{itemize}
    \end{column}

    \begin{column}{0.52\textwidth}
      \centering
      \includegraphics[width=\linewidth]{figures/imc_hierarchy_clean.png}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Overview\cite{rossenbach2025}}
  \begin{columns}[c]
    \begin{column}{0.70\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{1.5pt}
        \item A \emph{memristor} is an analog computing device with a programmable resistance states.
        \item Enables energy-efficient implementation of neural networks.
              \begin{itemize}
                \item Vector-matrix-multiplication (VMM) in $\mathcal{O}(1)$
              \end{itemize}
        \item Limitations:
              \begin{itemize}
                \item High analog inaccuracies, and thus need specific restrictions on mathematical operations.
                \item Production is still in early stages, experiments are on simulated hardware \cite{synatogenhenne}.
              \end{itemize}
      \end{itemize}
    \end{column}

    \begin{column}{0.30\textwidth}

    \end{column}
  \end{columns}
\end{frame}

\section{Motivation}
\begin{frame}{Motivation}
  \begin{itemize}
    \setlength{\itemsep}{10pt}
    \item CTC-based Conformer ASR has been shown to run on (simulated)
          memristor hardware \cite{rossenbach2025memristor}.
    \item How do \emph{transducer} models, with their added complexity compared to CTC, behave on memristor hardware?
          \begin{itemize}
            \item Autoregressive prediction network; internal language model (ILM).
            \item Joint network evaluated repeatedly per emitted label.
          \end{itemize}
    \item The impact of deploying various components of the model to the
          device (encoder only vs.\ encoder + prediction network vs.\ full model).
    \item \emph{Tradeoff}: more offloading $\Rightarrow$ more energy efficiency, but also more noise.
    \item Performance degradation across search paradigms -
          \begin{itemize}
            \item Lexicon-free vs.\ tree search.
            \item ILM and prior scaling - How does device noise change the optimal LM weighting?
          \end{itemize}
          % \item LM point of view: how does device deployment change the interplay between
          %       prior/ILM correction and the external LM?
  \end{itemize}
\end{frame}

\section{Prerequisites}

\begin{frame}{Analog In Memory Computing}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Multiplication computation done via Ohm's Law:
          \begin{equation}
            I = \frac{V}{R}
          \end{equation}
    \item Voltage $V$, current $I$ and resistance $R$.
    \item Possible to rewrite resistance as conductance $G$:
          \begin{equation}
            I = G V \quad \text{with} \quad G = \frac{1}{R}
          \end{equation}
    \item Addition done via Kirchhoff's Law for any connected node in a circuit:
          \begin{equation}
            \sum_{n=0}^{N} I_n = 0 \;\rightarrow\; I_0 = - \sum_{n=1}^{N} I_n
          \end{equation}
    \item Currents can be trivially added by simply connecting electric lines.
  \end{itemize}
\end{frame}

\begin{frame}{The ``Memristor''}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \textbf{Program weights by setting the resistance}:
      \begin{itemize}
        \setlength{\itemsep}{3pt}
        \item Apply $-V > -V_S \;\rightarrow$ set to low resistance
        \item Apply $\phantom{-}V > \phantom{-}V_R \;\rightarrow$ set to high resistance
        \item Intermediate states possible
        \item State is physically stored (non-volatile memory)
      \end{itemize}
      \vspace{0.4cm}
      \textbf{Problems}:
      \begin{itemize}
        \setlength{\itemsep}{3pt}
        \item Resulting resistance state varies with each programming cycle
        \item Non-linear relation of current and voltage
      \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth]{figures/memristor_iv_curve.png}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Memristor Array (1)}
  \begin{columns}[c]
    \begin{column}{0.54\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{4pt}
        \item Memristors as connecting element between vertical and horizontal circuit lines
        \item Each memristor is a weight of a matrix
        \item Column sum via simple line connection
        \item Weight matrices are typically fixed during application
      \end{itemize}
      \vspace{0.3cm}
      \textbf{Problems}:
      \begin{itemize}
        \setlength{\itemsep}{3pt}
        \item Only positive, non-zero weights possible
        \item Current leakage prohibits a true zero
      \end{itemize}
    \end{column}

    \begin{column}{0.46\textwidth}
      \centering
      \includegraphics[width=\linewidth]{figures/memristor_array_1.png}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Memristor Array (2)}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{2pt}
        \item Use positive and negative bitlines to create memristor pairs
        \item Each row-wise pair of memristors represents a weight
        \item[$\rightarrow$] Differential pair allows for negative and zero weight values:
              \begin{equation*}
                I_j = \sum_{i=1}^{N} V_i G_{i,j}^{+} - \sum_{i=1}^{N} V_i G_{i,j}^{-}
              \end{equation*}
      \end{itemize}
      \vspace{0.15cm}
      \textbf{Problems}:
      \begin{itemize}
        \setlength{\itemsep}{2pt}
        \item Cannot set paired memristors to identical conductances
        \item[$\rightarrow$] Results close to zero, but not guaranteed
      \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth]{figures/memristor_array_2.png}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{From Single Memristor to Layer Module: Concepts}
  \begin{itemize}
    \setlength{\itemsep}{8pt}
    \item \textbf{Multiple concepts to implement mapping}:
          \begin{itemize}
            \setlength{\itemsep}{6pt}
            \item \textbf{Tiling} of weight matrix:
                  \begin{itemize}
                    \item Arbitrary dimensions by tiling into smaller fixed-size arrays
                    \item E.g., $512 \times 512$ matrix into $16$ crossbar arrays of size $128 \times 128$
                  \end{itemize}
            \item \textbf{Slicing} of weight values to memristor states:
                  \begin{itemize}
                    \item Currently binary setting (high vs.\ low resistance)
                    \item Stacking of memristors for each magnitude bit ($k$-bit weight requires $k-1$ arrays)
                  \end{itemize}
            \item \textbf{Pairing} of positive ($+$) and negative ($-$) bitlines:
                  \begin{itemize}
                    \item Differential pairs represent signed matrix entries
                  \end{itemize}
          \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Non-Idealities and Synaptogen}
  \begin{itemize}
    \setlength{\itemsep}{5pt}
    \item Ideal crossbars are assumed by many simulators; real devices exhibit:
          \begin{itemize}
            \item Device-to-device and cycle-to-cycle programming variability
            \item Thermal (Johnson--Nyquist) noise
            \item Finite DAC/ADC resolution
            \item Imprecise intermediate (non-binary) conductance levels
          \end{itemize}
    \item Synaptogen \cite{hennen2024synaptogen}: generative device model based on a
          vector autoregression, fitted to 3M+ measured programming cycles of
          130\,nm memristors (STMicroelectronics).
    \item In our experiments (PyTorch, SynaptogenML): linear layers are replaced by
          crossbar simulation, the rest of the network stays conventional.
  \end{itemize}
\end{frame}

\begin{frame}{Quantization-Aware Training}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Memristor execution effectively means: low-precision (stacked binary) weights
          + 8-bit DAC/ADC on activations.
    \item Post-training quantization (PTQ) collapses at low precision;
          quantization-aware training (QAT) \cite{gholami2021survey} inserts
          fake quantization into the training from epoch 1.
          \begin{itemize}
            \item Symmetric per-tensor quantization, zero-point fixed to 0
                  (required by the pairwise encoding).
          \end{itemize}
  \end{itemize}
  \vfill

\section{Experiments}
\begin{frame}{Experimental Setup}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Data: LibriSpeech \cite{Librispeech} (960\,h), BPE subword targets
          (128 merges $\to$ 184 labels).
    \item Features: 80 log-mel filters, 10\,ms frame shift; SpecAugment during training.
    \item Encoder (shared by all models): 12-layer Conformer
          \cite{Gulati+Qin+:2020conformer}, 512-dim.
    \item Model types:
          \begin{enumerate}
            \item \emph{CTC} \cite{graves2006connectionist}: encoder only.
            \item \emph{FFNN-transducer}: prediction network is a 2-layer feed-forward
                  net (640-dim, tanh) over the previous-label embedding.
            \item \emph{Full-context transducer} \cite{graves2012sequence}: LSTM
                  prediction network (standard RNN-T).
          \end{enumerate}
    \item Quantization: QAT, per-tensor symmetric, e.g.\ 4-bit weights / 8-bit
          activations; zero-point 0.
    \item Stages: (1) FP32 baseline $\to$ (2) QAT (fake quantization) $\to$
          (3) memristor execution (Synaptogen crossbar simulation, multiple
          re-programmings per run).
  \end{itemize}
\end{frame}

\begin{frame}{Experiment Matrix}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Full matrix: 3 (model types) $\times$ 5 (search types) $\times$ 3 (stages).
    \item Search types:
          \begin{enumerate}
            \item Lexicon-free, no LM
            \item Lexicon-free + BPE LSTM LM
            \item Prefix-tree beam search, no LM
            \item Prefix-tree beam search + 4-gram LM
            \item Prefix-tree beam search + BPE LSTM LM
          \end{enumerate}
          (+ large-beam tree variants, beam 1024/2048)
    \item Prior/ILM correction and external LM scales are swept at decoding time.
    \item Reporting:
          \begin{itemize}
            \item Primary tables: one per model type (search types $\times$ stages).
            \item Appendix: tables across models at the same stage, for easy comparison.
          \end{itemize}
    \item Expectation: device deployment changes the interplay between prior/ILM
          correction and the LM --- quantization and noise reshape the acoustic
          scores, so the optimal LM weighting should shift.
  \end{itemize}
\end{frame}
