\begin{frame}{Overview \cite{rossenbach2025memristor}}
  \begin{columns}[c]
    \begin{column}{0.70\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{1.5pt}
        \item A \emph{memristor} is an analog computing device with a programmable resistance state \cite{strukov2008MM}.
        \item Enables fast, energy-efficient implementation of Vector-Matrix Multiplication (VMM) \cite{wan2022compute} -
              \begin{itemize}
                \item[$\rightarrow$] and thus neural networks.
              \end{itemize}
        \item Limitations:
              \begin{itemize}
                \item High analog inaccuracies, and thus need specific restrictions on mathematical operations \cite{wouters2016reliability}.
                \item Production is still in early stages \cite{infineon2022rram}.
                \item Current experiments are simulated using SynaptogenML \cite{hennen2024synaptogen} \cite{rossenbach2025memristor}.
              \end{itemize}
      \end{itemize}
    \end{column}

    \begin{column}{0.30\textwidth}

\end{frame}

\begin{frame}{Analog In Memory Computing \cite{aguirre2024hardware}}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Multiplication computation done via Ohm's Law:
          \begin{equation}
            V = IR
          \end{equation}
    \item Voltage $V$, current $I$ and resistance $R$.
    \item Possible to rewrite resistance as conductance $G$:
          \begin{equation}
            I = G V \quad \text{with} \quad G = \frac{1}{R}
          \end{equation}
    \item Addition done via Kirchhoff's Law for any connected node in a circuit:
          \begin{equation}
            \sum_{n=0}^{N} I_n = 0 \;\rightarrow\; I_0 = - \sum_{n=1}^{N} I_n
          \end{equation}
    \item Currents can be trivially added by simply connecting electric lines.
  \end{itemize}
\end{frame}

\begin{frame}{The Memristor \cite{hennen2024synaptogen}}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \emph{Key Idea}: Program weights by setting the resistance.
      \begin{itemize}
        \setlength{\itemsep}{3pt}
        \item Apply $-V > -V_S \;\rightarrow$ set to low resistance
        \item Apply $\phantom{-}V > \phantom{-}V_R \;\rightarrow$ set to high resistance
        \item Intermediate states possible, but difficult to program reliably
        \item State is physically stored (non-volatile memory)
      \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{figure}
        \centering
        \includegraphics[width=\linewidth]{figures/memristor_iv_curve.png}
        \caption{I-V characteristic curve of a memristor \cite{hennen2024synaptogen}}
      \end{figure}
    \end{column}
  \end{columns}
\end{frame}


\begin{frame}{Limitations \cite{rossenbach2025memristor}}
  \begin{itemize}
    \setlength{\itemsep}{3pt}
    \item[] \emph{Programming Limitations}
    \item Resulting resistance state varies with each programming cycle
    \item Non-linear relation of current and voltage
  \end{itemize}
  \hfill
  \begin{itemize}
    \setlength{\itemsep}{3pt}
    \item[] \emph{Cannot exactly model weights}
    \item Only positive, non-zero weights possible
    \item Current leakage prohibits a true zero
  \end{itemize}
\end{frame}

\begin{frame}{(Paired) Memristor Array \cite{aguirre2024hardware}}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{2pt}
        \item Use positive and negative bitlines to create memristor pairs
        \item Each row-wise pair of memristors represents a weight
        \item[$\rightarrow$] Differential pair allows for negative and zero weight values:
              \begin{equation*}
                I_j = \sum_{i=1}^{N} V_i G_{i,j}^{+} - \sum_{i=1}^{N} V_i G_{i,j}^{-}
              \end{equation*}
      \end{itemize}
      \vspace{2cm}
      \emph{Problems}:
      \begin{itemize}
        \setlength{\itemsep}{2pt}
        \item Cannot set paired memristors to identical conductances
        \item[$\rightarrow$] Difference close to zero, but not guaranteed
      \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{figure}
        \centering
        \includegraphics[width=\linewidth]{figures/memristor_array_2.png}
        \caption{Paired memristor array structure \cite{aguirre2024hardware}}
      \end{figure}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Memristor Neural Layers \cite{rossenbach2025memristor}}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item \emph{Tile} the weight matrix:
          \begin{itemize}
            \item Tile weight matrix with arbitrary dimensions into smaller fixed-size arrays.
            \item E.g., $512 \times 512$ weight matrix $\rightarrow$ $16$ crossbar arrays of size $128 \times 128$
          \end{itemize}
    \item \emph{Slice} weights into memristor states:
          \begin{itemize}
            \item Binary setting (high vs.\ low resistance)
            \item Stacking of memristors for each magnitude bit (\emph{$k$-bit} weight requires $k-1$ arrays)
          \end{itemize}
    \item \emph{Pair} positive ($+$) and negative ($-$) bitlines.
  \end{itemize}
\end{frame}

\begin{frame}{Quantization}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item The number of memristors required is proportional to the precision of the weights.
          \begin{itemize}
            \item Need to reduce the precision of the weights $\rightarrow$ quantization.
          \end{itemize}
    \item Quantization-Aware Training (QAT) peforms better than Post-training Quantization (PTQ), but requires more training time \cite{gholami2021survey}.
          \begin{itemize}
            \item[$\rightarrow$] High performance gap at low precision
          \end{itemize}
  \end{itemize}

\section{Motivation}
\begin{frame}{Motivation}
  \begin{itemize}
    \setlength{\itemsep}{10pt}
    \item CTC-based Conformer shown to run on (simulated) memristor hardware \cite{rossenbach2025memristor}.
    \item How do \emph{transducer} models, with their added complexity compared to CTC, behave on device?
          \begin{itemize}
            \item Autoregressive prediction network; internal language model (ILM).
            \item Joint network evaluated repeatedly per emitted label.
          \end{itemize}
    \item Performance degradation across search paradigms -
          \begin{itemize}
            \item Lexicon-free vs.\ tree search.
            \item ILM and prior scaling - How does device noise change the optimal LM weighting?
          \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Motivation}
  \begin{itemize}
    \setlength{\itemsep}{2pt}
    \item Performance impact of deploying components of the model on the device?
          \begin{itemize}
            \item Encoder only vs.\ encoder + prediction network vs.\ full model (encoder + prediction network + joint network)
          \end{itemize}
    \item \emph{Tradeoff}: more offloading $\Rightarrow$ more energy efficiency, but also more noise.
  \end{itemize}

\section{Experiments}
\begin{frame}{Experimental Setup}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Data: LibriSpeech \cite{Librispeech} (960\,h), BPE subword targets
          (128 merges $\to$ 184 labels).
    \item Features: 80 log-mel filters, 10\,ms frame shift; SpecAugment during training.
    \item Encoder: 12-layer Conformer \cite{Gulati+Qin+:2020conformer}, 512-dim.
    \item Model types:
          \begin{enumerate}
            \item \emph{CTC} \cite{graves2006connectionist}: encoder only.
            \item \emph{FFNN-transducer} \cite{graves2012sequence}: prediction network is a 2-layer feed-forward
                  net (640-dim) over the previous-label embedding; monotonic alignments \cite{tripathi2019monotonic}.
            \item \emph{Full-context transducer} \cite{graves2012sequence}: LSTM
                  prediction network; monotonic alignments \cite{tripathi2019monotonic}.
          \end{enumerate}
    \item Quantization: QAT, per-tensor symmetric, 8 and 4-bit weights, 8-bit activations; zero-point 0.
    \item Stages: (1) Full precision baseline $\to$ (2) QAT (fake quantization) $\to$
          (3) memristor simulation (3-5 runs).
  \end{itemize}
\end{frame}

\section{Future Work}

\begin{frame}{Future Work: Error Analysis}
  \begin{itemize}
    \setlength{\itemsep}{10pt}
    \item \emph{Error Analysis}:
          \begin{itemize}
            \item Substitutions vs.\ insertions vs.\ deletions.
                  \begin{itemize}
                    \item[$\rightarrow$] Why does lexicon-free decoding degrade faster than tree search?
                  \end{itemize}
            \item Acoustic posterior distortion and score calibration with external LMs.
          \end{itemize}
    \item \emph{Generalization}:
          \begin{itemize}
            \item Evaluation and training on larger/out-of-domain datasets.
          \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Future Work: Search Spaces}
  \begin{itemize}
    \setlength{\itemsep}{10pt}
    \item \emph{Decoding Parameter Adjustments}:
          \begin{itemize}
            \item Explore penalties for blank emissions.
            \item Tune softmax temperature to sharpen or soften posterior peaks under noise.
          \end{itemize}
    \item \emph{Search Space vs.\ Model \& QAT Level}:
          \begin{itemize}
            \item Current search space is large $\rightarrow$ hurts Real-Time Factor (RTF).
            \item Study the impact of restricing the search space on WER and RTF.
                  \begin{itemize}
                    \item[$\rightarrow$] How many more hypotheses are needed for on-device vs 4-bit vs.\ 8-bit vs.\ FP32 runs?
                    \item[$\rightarrow$] Compare hypothesis expansion across CTC and Transducer variants.
                  \end{itemize}
          \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Future Work: Robust Training \& Countermeasures}
  \begin{itemize}
    \setlength{\itemsep}{10pt}
    \item \emph{Noise-Aware QAT}:
          \begin{itemize}
            \item Inject simulated crossbar conductance noise directly into training forward pass.
          \end{itemize}
    \item \emph{Selective Layer Mapping}:
          \begin{itemize}
            \item Deal with noise in noise-sensitive components (e.g., joint network).
          \end{itemize}
    \item \emph{Denoising \& Output Compensation}:
          \begin{itemize}
            \item Post-processing to correct analog inaccuracies.
                  \begin{itemize}
                    \item[$\rightarrow$] Train a small neural network to denoise the memristor output.
                    \item[$\rightarrow$] Use a small calibration dataset to learn a linear transformation to correct the output.
                  \end{itemize}
          \end{itemize}
  \end{itemize}
\end{frame}
