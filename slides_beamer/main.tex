\section{Related Work}
\begin{frame}{Related Work}
  \begin{itemize}
    \setlength{\itemsep}{8pt}
    \item Memristive crossbars enable analog in-memory matrix--vector multiplication,
          promising energy-efficient inference for large neural networks
          \cite{strukov2008missing, aguirre2024hardware}.
    \item Most prior memristor ML studies are limited to small networks and tasks
          (MNIST-level, keyword spotting); common simulators
          (MemTorch, IBM AIHWKIT) rely on idealized hardware assumptions
          \cite{rossenbach2025memristor}.
    \item Rossenbach et al.\ \cite{rossenbach2025memristor} (Interspeech 2025):
          \begin{itemize}
            \item First full-scale ASR system (Conformer CTC, TED-LIUMv2) executed on
                  realistically simulated memristor hardware.
            \item Simulation via Synaptogen \cite{hennen2024synaptogen}, a device model
                  fitted to measured programming statistics.
            \item With QAT down to 3-bit weights: only $\sim$25\% relative WER
                  degradation on device.
          \end{itemize}
    \item Open questions left by \cite{rossenbach2025memristor}: transducer topologies
          and recurrent components on device --- the subject of this thesis.
  \end{itemize}
\end{frame}

\section{Motivation}
\begin{frame}{Motivation}
  \begin{itemize}
    \setlength{\itemsep}{10pt}
    \item Starting point: CTC-based Conformer ASR has been shown to run on (simulated)
          memristor hardware \cite{rossenbach2025memristor}.
    \item Curiosity: how does a \emph{transducer} model, architecturally more involved
          than CTC, behave on memristor hardware?
          \begin{itemize}
            \item Autoregressive prediction network with an internal language model (ILM).
            \item Joint network evaluated repeatedly per emitted label.
          \end{itemize}
    \item Study the impact of offloading various components of the transducer to the
          device (encoder only vs.\ encoder + prediction network).
    \item Study the actual impact of device noise, beyond pure quantization.
    \item Performance impact on lexicon-free vs.\ tree-based searches.
    \item LM point of view: how does device deployment change the interplay between
          prior/ILM correction and the external LM?
  \end{itemize}
\end{frame}

\section{Prerequisites}
\begin{frame}{Memristors and Crossbar Arrays}
  \begin{itemize}
    \setlength{\itemsep}{8pt}
    \item Memristor: passive two-terminal device with non-volatile, programmable
          resistance \cite{strukov2008missing}. A weight is stored as a conductance
          $G = 1/R$.
    \item Crossbar array: wordlines (rows) $\times$ bitlines (columns), one memristor
          per junction.
          \begin{itemize}
            \item Inputs applied as voltages $V_j$ on the wordlines.
            \item By Ohm's law and Kirchhoff's current law, each bitline aggregates a
                  column: \ $I_i = \sum_j G_{ij} V_j$.
            \item Matrix--vector multiplication happens in one physical step, in
                  parallel over all outputs.
          \end{itemize}
    \item Signed weights via pairwise encoding \cite{aguirre2024hardware}:
          $w_{ij} = G^{+}_{ij} - G^{-}_{ij}$ on two bitlines
          (an $N \times M$ matrix needs an $N \times 2M$ crossbar).
    \item Only binary programming is precise $\Rightarrow$ multi-bit weights via
          bit-stacking: a $k$-bit weight is split over $k{-}1$ stacked crossbars
          (4-bit: levels $-7, \ldots, 7$).
    \item Inputs/outputs pass 8-bit DAC/ADC converters (input range $\pm 0.6$\,V).
  \end{itemize}
\end{frame}

\begin{frame}{Non-Idealities and Synaptogen}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Ideal crossbars are assumed by many simulators; real devices exhibit:
          \begin{itemize}
            \item Device-to-device and cycle-to-cycle programming variability,
            \item Thermal (Johnson--Nyquist) noise,
            \item Finite DAC/ADC resolution,
            \item Imprecise intermediate (non-binary) conductance levels.
          \end{itemize}
    \item Synaptogen \cite{hennen2024synaptogen}: generative device model based on a
          vector autoregression, fitted to 3M+ measured programming cycles of
          130\,nm memristors (STMicroelectronics).
    \item In our experiments (PyTorch, SynaptogenML): linear layers are replaced by
          crossbar simulation, the rest of the network stays conventional.
  \end{itemize}
  \vfill

\begin{frame}{Quantization-Aware Training}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Memristor execution effectively means: low-precision (stacked binary) weights
          + 8-bit DAC/ADC on activations.
    \item Post-training quantization (PTQ) collapses at low precision;
          quantization-aware training (QAT) \cite{gholami2021survey} inserts
          fake quantization into the training from epoch 1.
          \begin{itemize}
            \item Symmetric per-tensor quantization, zero-point fixed to 0
                  (required by the pairwise encoding).
          \end{itemize}
  \end{itemize}
  \vfill

\begin{frame}{Model Topologies: What Can Run on Device?}
  \begin{columns}[t]
    \begin{column}{0.42\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{4pt}
        \small
        \item CTC \cite{graves2006connectionist}: encoder only, label-synchronous
              decoding, conditionally independent outputs.
        \item Transducer \cite{graves2012sequence}: encoder + prediction network +
              joint network, autoregressive over labels.
        \item All linear (matmul) layers map to crossbars (green).

        \item FFNN prediction network over the previous-label embedding: pure matmul

\section{Experiments}
\begin{frame}{Experimental Setup}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Data: LibriSpeech \cite{Librispeech} (960\,h), BPE subword targets
          (128 merges $\to$ 184 labels).
    \item Features: 80 log-mel filters, 10\,ms frame shift; SpecAugment during training.
    \item Encoder (shared by all models): 12-layer Conformer
          \cite{Gulati+Qin+:2020conformer}, 512-dim.
    \item Model types:
          \begin{enumerate}
            \item \emph{CTC} \cite{graves2006connectionist}: encoder only.
            \item \emph{FFNN-transducer}: prediction network is a 2-layer feed-forward
                  net (640-dim, tanh) over the previous-label embedding.
            \item \emph{Full-context transducer} \cite{graves2012sequence}: LSTM
                  prediction network (standard RNN-T).
          \end{enumerate}
    \item Quantization: QAT, per-tensor symmetric, e.g.\ 4-bit weights / 8-bit
          activations; zero-point 0.
    \item Stages: (1) FP32 baseline $\to$ (2) QAT (fake quantization) $\to$
          (3) memristor execution (Synaptogen crossbar simulation, multiple
          re-programmings per run).
  \end{itemize}
\end{frame}

\begin{frame}{Experiment Matrix}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Full matrix: 3 (model types) $\times$ 5 (search types) $\times$ 3 (stages).
    \item Search types:
          \begin{enumerate}
            \item Lexicon-free, no LM
            \item Lexicon-free + BPE LSTM LM
            \item Prefix-tree beam search, no LM
            \item Prefix-tree beam search + 4-gram LM
            \item Prefix-tree beam search + BPE LSTM LM
          \end{enumerate}
          (+ large-beam tree variants, beam 1024/2048)
    \item Prior/ILM correction and external LM scales are swept at decoding time.
    \item Reporting:
          \begin{itemize}
            \item Primary tables: one per model type (search types $\times$ stages).
            \item Appendix: tables across models at the same stage, for easy comparison.
          \end{itemize}
    \item Expectation: device deployment changes the interplay between prior/ILM
          correction and the LM --- quantization and noise reshape the acoustic
          scores, so the optimal LM weighting should shift.
  \end{itemize}
\end{frame}

\section{Future Work}
\begin{frame}{Future Work}
  \begin{itemize}
    \setlength{\itemsep}{10pt}
    \item Error analysis: which error types does device noise introduce or amplify?
    \item Countermeasures: denoising (e.g.\ MLP), search parameters, LM parameters,

    \item Fine-tuning of the (quantized) model; effect of noise introduction during
          training.
    \item Generalization: comparison across different datasets / data conditions.
  \end{itemize}
\end{frame}
