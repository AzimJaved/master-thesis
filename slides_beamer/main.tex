\section{Motivation}

\begin{frame}{Motivation}
  \begin{itemize}
    \setlength{\itemsep}{10pt}
    \item Memristor hardware enables fast, energy-efficient neural network inference.
          \begin{itemize}
            \item[$\rightarrow$] but introduces analog inaccuracies.
          \end{itemize}
    \item CTC-based Conformers have been evaluated on memristor hardware \cite{rossenbach2025memristor}.
    \item Transducers add two components, which are repeatedly invoked at inference:
          \begin{itemize}
            \item Predictor network
            \item Joiner network
          \end{itemize}
    \item \emph{Objective}:
          \begin{itemize}
            \item Characterize the effect of quantization and device noise on transducer ASR.
            \item Assess training and decoding countermeasures.
          \end{itemize}
  \end{itemize}
\end{frame}

\section{Prerequisites}

\begin{frame}{Overview}
  \begin{columns}[c]
    \begin{column}{0.70\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{1.5pt}
        \item A \emph{memristor} is an analog computing device with a programmable resistance state \cite{strukov2008MM}.
        \item Enables fast, energy-efficient implementation of Vector-Matrix Multiplication (VMM)
              \begin{itemize}
                \item[$\rightarrow$] and thus neural networks.
              \end{itemize}
              \emph{Key Points}:
              \begin{itemize}
                \item Introduces analog inaccuracies.
                \item Production is still in early stages.
                \item Experiments are simulated using SynaptogenML \cite{rossenbach2025memristor}.
              \end{itemize}
      \end{itemize}
    \end{column}

    \begin{column}{0.30\textwidth}
      \begin{figure}
        \centering
        \includegraphics[width=0.85\linewidth]{figures/resistor_memristor_clean.png}
        \caption{Resistor and memristor schematics}
      \end{figure}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Analog In-Memory Computing}
  \begin{columns}[c]
    \begin{column}{0.54\textwidth}
      \begin{minipage}[t][0.65\textheight][t]{\linewidth}
        \begin{itemize}
          \setlength{\itemsep}{4pt}
          \item Memristors serve as the connecting elements between vertical and horizontal lines.
          \item Each memristor's conductance is an element of a constant matrix.
          \item Column sum via line connection.
        \end{itemize}
        \vfill
        \emph{Key Points}:
        \begin{itemize}
          \item Cannot set a memristor to an exact conductance $\Rightarrow$ use high/low bit values.
        \end{itemize}
      \end{minipage}
    \end{column}

    \begin{column}{0.46\textwidth}
      \begin{figure}
        \centering
        \includegraphics[width=0.90\linewidth]{figures/memristor_array_1.png}
        \caption{Crossbar array performing analog VMM \cite{rossenbach2025memristor}}
      \end{figure}
    \end{column}
  \end{columns}
  \hfill
\end{frame}

\begin{frame}{Memristor Crossbar Layout}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{2pt}
        \item Use positive and negative bitlines to create memristor pairs
        \item Each row-wise pair of memristors represents a weight
        \item[$\rightarrow$] For $N$ input rows, with row index $i$ and output-column index $j$:
              \begin{equation*}
                I_j = \sum_{i=1}^{N} V_i G_{i,j}^{+} - \sum_{i=1}^{N} V_i G_{i,j}^{-}
              \end{equation*}
      \end{itemize}
      \vspace{0.5cm}
      \emph{Key Points}:
      \begin{itemize}
        \setlength{\itemsep}{2pt}
        \item Even with bit values, the programming is noisy.
        \item Further, there is noise when resultant values are read out.
      \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{figure}
        \centering
        \includegraphics[width=\linewidth]{figures/memristor_array_2.png}
        \caption{Paired memristor array structure \cite{aguirre2024hardware}}
      \end{figure}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Experimental Setup: Models and Quantization}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Model types:
          \begin{enumerate}
            \item \emph{Connectionist Temporal Classification (CTC)} \cite{graves2006connectionist}: encoder only.
            \item \emph{Feed Forward Neural Network (FFNN) transducer} \cite{graves2012sequence}: 2-layer, 640-dim feed-forward predictor.
                  \begin{itemize}
                    \item Conditions on the previous-label embedding.
                  \end{itemize}
            \item \emph{Full-context transducer} \cite{graves2012sequence}: LSTM predictor over label history.
          \end{enumerate}
    \item The joiner network has two layers.
    \item Both transducers use monotonic alignments \cite{tripathi2019monotonic}.
    \item \emph{Model sizes}:
          CTC \ctcParameterMillions  parameters; FFNN transducer \ffnnParameterMillions  parameters; Full-context transducer \lstmParameterMillions  parameters.
    \item \emph{Quantization}: Quantization-Aware Training (QAT).
    \item[] \begin{itemize}
            \item 8-/4-bit weights, 8-bit activations; zero-point 0.
            \item Per-tensor symmetric quantization.
          \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Experimental Setup: Training}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item \emph{Data}: LibriSpeech \cite{Librispeech}, 960\,h. 184 Byte-Pair Encoding (BPE) subword labels.
    \item \emph{Features}: 80 log-mel filters, 10\,ms frame shift; downsampling factor 4, SpecAugment during training.
    \item \emph{Encoder}: 12-layer Conformer \cite{Gulati+Qin+:2020conformer}, 512-dim.
    \item \emph{Training}: 100 epochs, RAdam optimizer, $\epsilon=10^{-12}$, weight decay = 0.01.
    \item[] \begin{itemize}
            \item OCLR: initial LR $7\times10^{-6}$; peak $5\times10^{-4}$ after 48 epochs; decay to $5\times10^{-5}$ over 48 epochs; final LR $1\times10^{-7}$ over 4 epochs.
          \end{itemize}
    \item \emph{Runs}:
          \begin{itemize}
            \item Full-precision baseline and from scratch training with QAT.
            \item Simulated memristor results use 3 (transducer) or 5 (CTC) runs. Results reported as mean WER (\%) $\pm$ std-dev. Some runs may use 1 cycle.
            \item The default analog-to-digital converter (ADC) config. uses 4 precision bits and 4 range bits (ADC: 4/4), except for the positional encoding: 1/7.
          \end{itemize}
    \item WER (\%) on LibriSpeech dev-other, for lex-free (greedy) decoding, tree search (beam size 1024), and tree search with a 4-gram LM (beam size 2048).
  \end{itemize}
\end{frame}

\section{Future Work}

\begin{frame}{Future Work}
  \begin{itemize}
    \setlength{\itemsep}{10pt}
    \item \emph{Generalization}:
          \begin{itemize}
            \item Evaluation and training on larger/out-of-domain datasets.
          \end{itemize}
    \item \emph{Countermeasures to the Joiner Breakdown}:
          \begin{itemize}
            \item Regularization during training to constrain outliers.
            \item Resetting observers after a few warmup steps during training.
                  \begin{itemize}
                    \item [$\rightarrow$] Use moving-average-based observers instead of the min-max observers.
                  \end{itemize}
            \item Alternative intermediate activation functions between the joiner layers.
          \end{itemize}
    \item \emph{Programming Corrections}
          \begin{itemize}
            \item Explore additional programming correction configurations and strategies.
                  \begin{itemize}
                    \item[$\rightarrow$] E.g., errors in the most significant bit are much more impactful.
                    \item[$\rightarrow$] Real-world devices may change characteristics due to heat generated over multiple programming cycles.
                    \item[$\rightarrow$] Various layers may have different sensitivities to programming errors.
                  \end{itemize}
          \end{itemize}

  \end{itemize}
\end{frame}

\begin{frame}{Future Work: Search}
  \begin{itemize}
    \setlength{\itemsep}{10pt}
    \item \emph{Decoder Parameters}:
          \begin{itemize}
            \item Explore label emission rewards and blank penalties on optimal cases.
            \item Tune softmax temperature to sharpen or soften the distribution under noise.
          \end{itemize}
    \item \emph{Search Space Analysis: Memristor \& QAT Level}:
          \begin{itemize}
            \item Larger search spaces require more energy.
            \item Study the impact of restricting the search space on WER.
                  \begin{itemize}
                    \item[$\rightarrow$] How many more hypotheses are needed for on-device (simulated) vs.\ 4-bit vs.\ 8-bit vs.\ full-precision runs?
                    \item[$\rightarrow$] Compare hypothesis expansion across CTC and transducer variants.
                  \end{itemize}
          \end{itemize}
    \item \emph{Search Error vs.\ Model Errors}:
          \begin{itemize}
            \item[$\rightarrow$] Rescore memristor-simulated hypotheses with the corresponding QAT model.
          \end{itemize}
    \item \emph{Search Paradigms}:
          \begin{itemize}
            \item Explore alternative search algorithms and language model combinations (e.g., lexicon-free with an LSTM/transformer-based LM).
          \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Future Work: Robust Training \& Countermeasures}
  \begin{itemize}
    \setlength{\itemsep}{10pt}
    \item \emph{Noise- and Quantization-Aware Training}:
          \begin{itemize}
            \item Inject simulated crossbar noise directly into the training pass.
            \item Regularization to encourage robustness to noise and prevent ADC saturation.
          \end{itemize}
    \item \emph{Denoising \& Output Corrections}:
          \begin{itemize}
            \item Calibrate memristor-simulated logits, using a small dataset
            \begin{itemize}
              \item[$\rightarrow$] with the standard loss function.
              \item[$\rightarrow$] with the corresponding QAT model as a teacher.
            \end{itemize}
            \item Train additional full-precision/quantized/memristor denoising layers with the memristor-simulated model.
            \item Add such denoising layers within the model, after certain layers.
            \item Explore whether the learnt denoising layers can be mapped across multiple models.
            \item Low-cost output corrections: scaling and blank-bias correction, or diagonal transformations with optional low-rank correction.
          \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Analog In-Memory Computing}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Multiplication computation via Ohm's law:
          \begin{equation}
            V = IR
          \end{equation}
    \item Voltage $V$, current $I$, and resistance $R$.
    \item Possible to rewrite resistance as conductance $G$:
          \begin{equation}
            I = G V \quad \text{with} \quad G = \frac{1}{R}
          \end{equation}
    \item Addition done via Kirchhoff's law:
          \begin{itemize}
            \item For any connected node in a circuit, with $N$ connected lines:
          \end{itemize}
          \begin{equation}
            \sum_{n=0}^{N} I_n = 0 \;\rightarrow\; I_0 = - \sum_{n=1}^{N} I_n
          \end{equation}
    \item Currents can be trivially added by simply connecting electric lines.
  \end{itemize}
\end{frame}

\begin{frame}{The Memristor}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \emph{Key Idea}: Encode each weight bit as a high- or low-resistance state.
      \begin{itemize}
        \setlength{\itemsep}{3pt}
        \item Let $V_S$ be the set voltage and $V_R$ the reset voltage thresholds.
        \item Apply $V < V_S \;\rightarrow$ set to low resistance.
        \item Apply $V > V_R \;\rightarrow$ set to high resistance.
        \item Intermediate states are difficult to program reliably
              for the devices measured in Synaptogen.
        \item Memristor state is non-volatile.
      \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{figure}
        \centering
        \includegraphics[width=\linewidth]{figures/memristor_iv_curve.png}
        \caption{I{--}V characteristic curve of a memristor \cite{hennen2024synaptogen}}
        {\scriptsize $R_L/R_H$: low-/high-resistance state; $I_L(V)/I_H(V)$: corresponding current; $V_{\max}$: maximum applied voltage; $\mathrm{RESET}(V)$: current during the reset transition.}
      \end{figure}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Quantization}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item The number of memristors required is proportional to the precision of the weights.
          \begin{itemize}
            \item Need to reduce the precision of the weights $\rightarrow$ quantization.
          \end{itemize}
    \item Quantization-Aware Training (QAT) performs better than Post-Training Quantization (PTQ), but requires more training time \cite{gholami2021survey}.
          \begin{itemize}
            \item[$\rightarrow$] High performance gap at low precision
          \end{itemize}
  \end{itemize}

\begin{frame}{Memristor Neural Layers \cite{rossenbach2025memristor}}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item \emph{Tile} the weight matrix:
          \begin{itemize}
            \item Tile weight matrix with arbitrary dimensions into smaller fixed-size arrays.
            \item E.g., $512 \times 512$ weight matrix $\rightarrow$ $16$ crossbar arrays of size $128 \times 128$.
          \end{itemize}
    \item \emph{Slice} weights into memristor states:
          \begin{itemize}
            \item Binary setting (high vs.\ low resistance).
            \item Stacking of memristors for each magnitude bit (\emph{$k$-bit} weight requires $k-1$ arrays).
          \end{itemize}
    \item \emph{Pair} positive ($+$) and negative ($-$) bitlines.
  \end{itemize}
\end{frame}
